const FLAGS = {
  en: "🇬🇧", ru: "🇷🇺", ar: "🇦🇪", uk: "🇺🇦", fr: "🇫🇷", de: "🇩🇪", es: "🇪🇸",

/* ============================================================================
   Menus_iconUpdateBtn — icon-only update btn for a single field.
   Mode lifecycle:
     idle    -> click pencil  -> mode='editing'
     editing -> click check   -> requestConfirm(label, prev, next)
     editing -> click x       -> mode='idle'
============================================================================ */
function Menus_iconUpdateBtn({ tooltip, icon, active, disabled, onClick }) {
  return (
    <button
      type="button"
      className={`menus_iconUpdateBtn ${active ? "active" : ""}`}
      data-tooltip={tooltip}
      aria-label={tooltip}
      aria-disabled={disabled ? "true" : undefined}
      disabled={disabled}
      onClick={disabled ? undefined : onClick}>
      {icon || I.pencil}
    </button>
  );
}
window.Menus_iconUpdateBtn = Menus_iconUpdateBtn;

/* ============================================================================
   Menus_breadcrumb — clickable path replacing "Back to Menus".
   trail: [{key, label, onClick?}]; the last crumb is the current page.
============================================================================ */
function Menus_breadcrumb({ trail }) {
  return (
    <nav className="menus_breadcrumb" aria-label="Breadcrumb">
      {trail.map((c, i) => {
        const isLast = i === trail.length - 1;
        return (
          <Fragment key={c.key}>
            <button
              type="button"
              className={`menus_breadcrumb_crumb ${isLast ? "current" : ""}`}
              onClick={isLast ? undefined : c.onClick}
              aria-current={isLast ? "page" : undefined}>
              {c.label}
            </button>
            {!isLast && <span className="menus_breadcrumb_sep">/</span>}
          </Fragment>
        );
      })}
    </nav>
  );
}
window.Menus_breadcrumb = Menus_breadcrumb;

/* ============================================================================
   Menus_translations — expandable panel showing per-language values.
   data: object of { langCode: string } | string per kind ("short"/"long").
============================================================================ */
function Menus_translations({ title, data, readOnly = true, multiline = false }) {
  const entries = Object.entries(data || {}).filter(([k, v]) => v != null && v !== "");
  if (!entries.length) {
    return (
      <div className="menus_translations">
        <p className="menus_translations_title">{title}</p>
        <p style={{ margin: 0, fontSize: 12, color: "var(--menus-text-faint)" }}>No translations recorded.</p>
      </div>
    );
  }
  return (
    <div className="menus_translations">
      <p className="menus_translations_title">{title}</p>
      {entries.map(([lang, value]) => (
        <div key={lang} className="menus_translations_row">
          <span className="menus_translations_flag" title={lang.toUpperCase()}>
            {FLAGS[lang] || LANG_LABEL[lang] || lang.toUpperCase()}
          </span>
          {multiline ? (
            <textarea className="menus_translations_textarea" defaultValue={value} readOnly={readOnly} rows={3} />
          ) : (
            <input className="menus_translations_input" defaultValue={value} readOnly={readOnly} />
          )}
          <span className="menus_translations_code">{lang}</span>
        </div>
      ))}
    </div>
  );
}
window.Menus_translations = Menus_translations;

/* ============================================================================
   Menus_dropZone — drag-and-drop file drop zone (no real upload — demo only).
============================================================================ */
function Menus_dropZone({ label = "Drop a file here, or click to browse", hint, disabled, onFile }) {
  const [over, setOver] = _useState(false);
  const inputRef = _useRef(null);
  return (
    <label
      className={`menus_dropZone ${over ? "menus_dropZone--over" : ""} ${disabled ? "menus_dropZone--disabled" : ""}`}
      onDragOver={(e) => { if (disabled) return; e.preventDefault(); setOver(true); }}
      onDragLeave={() => setOver(false)}
      onDrop={(e) => {
        if (disabled) return;
        e.preventDefault();
        setOver(false);
        const f = e.dataTransfer.files?.[0];
        if (f && onFile) onFile(f);
      }}>
      <span className="menus_dropZone_icon">{I.upload}</span>
      <span className="menus_dropZone_label">{label}</span>
      {hint && <span className="menus_dropZone_hint">{hint}</span>}
      <input
        ref={inputRef}
        type="file"
        accept="image/*,.pdf,.docx,.xlsx,.txt"
        disabled={disabled}
        onChange={(e) => {
          const f = e.target.files?.[0];
          if (f && onFile) onFile(f);
        }}
      />
    </label>
  );
}
window.Menus_dropZone = Menus_dropZone;

/* ============================================================================
   Menus_filePreview — image thumbnail or icon for non-image files.
============================================================================ */
function Menus_filePreview({ file, label }) {
  if (!file) {
    return (
      <div className="menus_filePreview" title="No file">
        <span className="menus_filePreview_thumb" style={{ opacity: 0.5 }}>{I.file}</span>
        <span className="menus_filePreview_label">—</span>
      </div>
    );
  }
  // file can be: { path, fileType, sizeInBytes, ref } or just a URL string
  const isStr = typeof file === "string";
  const path = isStr ? file : file.path;
  const type = isStr ? "image" : (file.fileType || "").toLowerCase();
  const isImg = type === "image" || /\.(png|jpe?g|webp|gif|svg)$/i.test(path || "");

  const iconForType = {
    pdf: I.pdf, docx: I.word, doc: I.word, xlsx: I.excel, xls: I.excel,
  }[type] || I.file;

  return (
    <div className="menus_filePreview" title={isStr ? path : (file.description || file.ref)}>
      <span className="menus_filePreview_thumb">
        {isImg && path ? <img src={path} alt={label || (file.ref || "")} /> : iconForType}
      </span>
      <span className="menus_filePreview_label">{label || (isStr ? "file" : (file.ref || type || "file"))}</span>
      {!isStr && file.sizeInBytes != null && (
        <span className="menus_filePreview_meta">{window.formatBytes(file.sizeInBytes)}</span>
      )}
    </div>
  );
}
window.Menus_filePreview = Menus_filePreview;

/* ============================================================================
   Menus_confirmModal — "are you sure" double-confirm modal.
============================================================================ */
function Menus_confirmModal({ open, title, subtitle, prevValue, nextValue, fieldLabel, danger, onCancel, onConfirm }) {
  _useEffect(() => {
    if (!open) return;
    const onKey = (e) => {
      if (e.key === "Escape") onCancel?.();
      if (e.key === "Enter") onConfirm?.();
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [open, onCancel, onConfirm]);

  if (!open) return null;

  const showDiff = prevValue !== undefined || nextValue !== undefined;
  return (
    <div className="menus_confirmModal_overlay" onClick={onCancel} role="dialog" aria-modal="true">
      <div className="menus_confirmModal" onClick={(e) => e.stopPropagation()}>
        <header className="menus_confirmModal_header">
          <div>
            {subtitle && <p className="menus_confirmModal_subtitle">{subtitle}</p>}
            <h2 className="menus_confirmModal_title">{title || "Are you sure?"}</h2>
          </div>
          <button type="button" className="menus_confirmModal_close" onClick={onCancel} aria-label="Close">
            {I.x}
          </button>
        </header>
        <div className="menus_confirmModal_body">
          <p>
            {fieldLabel ? (
              <>You are about to update <strong>{fieldLabel}</strong>. This change cannot be undone.</>
            ) : (
              <>This change cannot be undone.</>
            )}
          </p>
          {showDiff && (
            <div className="menus_confirmModal_diff">
              <div className="menus_confirmModal_diff_col prev">
                <span className="menus_confirmModal_diff_col_label">Current</span>
                <span className="menus_confirmModal_diff_col_value">{String(prevValue ?? "—")}</span>
              </div>
              <div className="menus_confirmModal_diff_col next">
                <span className="menus_confirmModal_diff_col_label">New</span>
                <span className="menus_confirmModal_diff_col_value">{String(nextValue ?? "—")}</span>
              </div>
            </div>
          )}
        </div>
        <footer className="menus_confirmModal_actions">
          <button type="button" className="menus_confirmModal_btn" onClick={onCancel}>Cancel</button>
          <button type="button" className={`menus_confirmModal_btn ${danger ? "danger" : "primary"}`} onClick={onConfirm}>
            {danger ? "Yes, delete" : "Yes, confirm"}
          </button>
        </footer>
      </div>
    </div>
  );
}
window.Menus_confirmModal = Menus_confirmModal;

/* ============================================================================
   Menus_fieldRow — generic in-place field editor with icon update + cancel.
   The component is *display-only*; commit goes through props.onRequestConfirm
   so the parent can pop the confirm modal.
============================================================================ */
function Menus_fieldRow({
  label,           // shown above input
  value,           // current value
  type = "text",   // "text" | "number" | "textarea" | "select"
  options,         // for select
  fieldKey,        // logical key for tracking edit
  lockedBy,        // global "isUpdatingAll" disables the per-field button
  editingField,    // which field is currently being edited (or null)
  setEditingField, // setter
  onRequestConfirm,// (label, prev, next) => void  — pop modal
  multiline,
  tooltip,
  extraControls,   // node — rendered next to the update icon (e.g. translations toggle)
  inputProps = {},
}) {
  const isEditing = editingField === fieldKey;
  const [draft, setDraft] = _useState(value);

  _useEffect(() => { setDraft(value); }, [value, isEditing]);

  const disabledByOther = (!!editingField && !isEditing) || (lockedBy && !isEditing);

  const startEdit = () => setEditingField(fieldKey);
  const cancelEdit = () => { setDraft(value); setEditingField(null); };
  const saveEdit = () => {
    if (draft === value) { setEditingField(null); return; }
    onRequestConfirm?.(label, value, draft, () => setEditingField(null));
  };

  const Input = type === "textarea" || multiline ? "textarea" : (type === "select" ? "select" : "input");
  const inputClass = type === "textarea" || multiline
    ? "menus_menuItem_view_one_topRight_description_textarea"
    : "menus_menuItem_view_one_topRight_name_input";

  const inputEl = type === "select" ? (
    <select className={inputClass} value={draft ?? ""} disabled={!isEditing}
      onChange={(e) => setDraft(e.target.value)}>
      {(options || []).map((o) => <option key={o} value={o}>{o}</option>)}
    </select>
  ) : (
    <Input
      type={type === "number" ? "number" : "text"}
      className={inputClass}
      value={draft ?? ""}
      readOnly={!isEditing}
      onChange={(e) => setDraft(type === "number" ? Number(e.target.value) : e.target.value)}
      rows={multiline ? 4 : undefined}
      {...inputProps}
    />
  );

  return (
    <div className="menus_menuItem_view_one_topRight_name" style={{ width: "100%" }}>
      <div className="menus_menuItem_view_one_topRight_name_label">
        <label>{label}</label>
        <div className="menus_menuItem_view_one_topRight_controlls">
          {extraControls}
          {isEditing ? (
            <>
              <Menus_iconUpdateBtn icon={I.check} tooltip="Confirm" onClick={saveEdit} active />
              <Menus_iconUpdateBtn icon={I.x} tooltip="Cancel" onClick={cancelEdit} />
            </>
          ) : (
            <Menus_iconUpdateBtn icon={I.pencil} tooltip={tooltip || `Update ${label}`} onClick={startEdit} disabled={disabledByOther} />
          )}
        </div>
      </div>
      {inputEl}
    </div>
  );
}
window.Menus_fieldRow = Menus_fieldRow;
