        />

        <p>Updated By: -- {menu.updatedBy}</p>
        <p>Updated At: -- {window.formatDate(menu.updatedAt)}</p>

          if (!cat) return null;

          return (

                    <span className="menus_category_header_menuItems_count">{items.length}</span>
                    <span className="menus_category_header_menuItems_label">Menu Items</span>
                  </p>
                </aside>
                <div className="menus_category_header_timing">
                  {cat.activeTimings.isAlwaysActive ? <p className="menus_category_header_alwaysActive">Always Active</p> : (
                    cat.activeTimings.windows.map((w, i) => (
                      <div key={i}>
                        <p className="menus_category_header_windowLabel">{w.label}</p>
                        <span className="menus_category_header_windowTime_from">{w.from}</span>–

                <aside className="menus_category_header_actions">
                  <button className="menus_category_header_button edit">Edit</button>
                  <button className="menus_category_header_button dropdown">More</button>
                </aside>
              </div>
              <div className="menus_categories_menuItems">
                {items.length ? (
                  <Menus_menuItems_table

                    expandedId={expandedItemId} setExpandedId={setExpandedItemId}
                  />
                ) : <div className="menus_empty_menuItems"><p>No items in this category yet.</p></div>}
              </div>
            </div>
          );

      </div>

  );

/* ============================================================================
   Menus_form — creation form (ownerType pick -> label/title input -> submit)
   Reused for all five entity types.
============================================================================ */
function Menus_form({ kind, onClose, onCreate }) {
  const [step, setStep] = _aState(0);
  const [ownerType, setOwnerType] = _aState("brand");
  const [label, setLabel] = _aState("");
  const labels = {
    menu: { title: "Create a menu", input: "Menu label", hint: "e.g. \"Main Dining Menu\"" },
    menuItem: { title: "Create a menu item", input: "Item name", hint: "e.g. \"Strawberry Pavlova\"" },
    modifier: { title: "Create a modifier", input: "Modifier title", hint: "e.g. \"Sauce\"" },
    option: { title: "Create an option", input: "Option name", hint: "e.g. \"Strawberry coulis\"" },
    category: { title: "Create a category", input: "Category name", hint: "e.g. \"Desserts\"" },
  }[kind] || labels.menu;

  const steps = ["Owner", "Name", "Confirm"];
  const canNext = step === 0 ? !!ownerType : (step === 1 ? label.trim().length > 1 : true);

  return (
    <div className="menus_confirmModal_overlay" role="dialog" aria-modal="true">
      <div className="menus_confirmModal" style={{ width: "min(640px, 100%)" }}>
        <header className="menus_confirmModal_header">
          <div>
            <p className="menus_confirmModal_subtitle">Create</p>
            <h2 className="menus_confirmModal_title">{labels.title}</h2>
          </div>
          <button type="button" className="menus_confirmModal_close" onClick={onClose} aria-label="Close">{I.x}</button>
        </header>
        <div className="menus_confirmModal_body">
          <div className="menus_form">
            <div className="menus_form_steps">
              {steps.map((s, i) => (
                <div key={s} className={`menus_form_step ${step === i ? "active" : (i < step ? "done" : "")}`}>
                  <span className="menus_form_step_num">{i + 1}</span>{s}
                </div>
              ))}
            </div>

            {step === 0 && (
              <div className="menus_form_field">
                <label>Owner type</label>
                <p className="menus_form_helper">Pick whether this {kind} belongs to your brand or to a competitor.</p>
                <div className="menus_form_ownerTypePicker">
                  <button type="button" className={`menus_form_ownerType_card ${ownerType === "brand" ? "selected" : ""}`} onClick={() => setOwnerType("brand")}>
                    {I.store}
                    <span className="menus_form_ownerType_label">Brand</span>
                    <span className="menus_form_ownerType_hint">Owned by your kitchen.</span>
                  </button>
                  <button type="button" className={`menus_form_ownerType_card ${ownerType === "competitor" ? "selected" : ""}`} onClick={() => setOwnerType("competitor")}>
                    {I.rival}
                    <span className="menus_form_ownerType_label">Competitor</span>
                    <span className="menus_form_ownerType_hint">External — for tracking and competition.</span>
                  </button>
                </div>
              </div>
            )}

            {step === 1 && (
              <div className="menus_form_field">
                <label>{labels.input}</label>
                <input className="menus_form_input" autoFocus value={label} onChange={(e) => setLabel(e.target.value)} placeholder={labels.hint} />
                <p className="menus_form_helper">You can fill in all other details after the entry exists.</p>
              </div>
            )}

            {step === 2 && (
              <div className="menus_form_field">
                <label>Review</label>
                <div className="menus_confirmModal_diff">
                  <div className="menus_confirmModal_diff_col">
                    <span className="menus_confirmModal_diff_col_label">Owner</span>
                    <span className="menus_confirmModal_diff_col_value">{ownerType}</span>
                  </div>
                  <div className="menus_confirmModal_diff_col next">
                    <span className="menus_confirmModal_diff_col_label">{labels.input}</span>
                    <span className="menus_confirmModal_diff_col_value">{label}</span>
                  </div>
                </div>
                <p className="menus_form_helper">Files, images, pricing, modifiers etc. are added later via the view session.</p>
              </div>
            )}
          </div>
        </div>
        <footer className="menus_confirmModal_actions">
          <div className="menus_form_actions_left">
            {step > 0 && <button className="menus_confirmModal_btn" onClick={() => setStep(step - 1)}>Back</button>}
          </div>
          <div className="menus_form_actions_right">
            <button className="menus_confirmModal_btn" onClick={onClose}>Cancel</button>
            {step < 2
              ? <button className="menus_confirmModal_btn primary" disabled={!canNext} onClick={() => setStep(step + 1)}>Next</button>
              : <button className="menus_confirmModal_btn primary" onClick={() => onCreate({ kind, ownerType, label })}>Create</button>}
          </div>
        </footer>
      </div>
    </div>
  );
}

/* ============================================================================
   MAIN — session app. Holds all state for session/viewingType/selected/etc.
============================================================================ */
function MenusApp() {
  const [session, setSession] = _aState("menus");           // menus | items | modifiers | options
  const [viewingType, setViewingType] = _aState("all");     // all | single
  const [selectedId, setSelectedId] = _aState(null);        // id of the open menu/item/modifier/option
  const [ownerType, setOwnerType] = _aState("both");        // brand | competitor | both
  const [editingField, setEditingField] = _aState(null);    // per-field edit mode
  const [isUpdatingAll, setIsUpdatingAll] = _aState(false); // top-level update mode
  const [confirm, setConfirm] = _aState(null);              // {title, prev, next, fieldLabel, onCommit}
  const [showForm, setShowForm] = _aState(null);            // null | menu|menuItem|modifier|option|category
  const [expandedRowId, setExpandedRowId] = _aState(null);
  const [toast, setToast] = _aState(null);

  _aEffect(() => {
    if (!toast) return;
    const id = setTimeout(() => setToast(null), 2400);
    return () => clearTimeout(id);
  }, [toast]);

  /* ---- Derived: breadcrumb trail ---- */
  const trail = _aMemo(() => {
    const sessLabel = { menus: "Menus", items: "Items", modifiers: "Modifiers", options: "Options" }[session];
    const t = [{ key: "menus", label: "Menus", onClick: () => goto("menus", "all", null) }];
    if (session !== "menus") {
      t.push({ key: session, label: sessLabel, onClick: () => goto(session, "all", null) });
    }
    if (viewingType === "single" && selectedId) {
      const entity = (session === "menus" && MOCK.menus.find((m) => m._id === selectedId))
        || (session === "items" && MOCK.items[selectedId])
        || (session === "modifiers" && MOCK.modifiers[selectedId])
        || (session === "options" && MOCK.options[selectedId]);
      const lbl = entity?.label || entity?.name?.label || entity?.title?.label || "Detail";
      t.push({ key: "selected", label: lbl, onClick: undefined });
    }
    return t;
  }, [session, viewingType, selectedId]);

  function goto(nextSession, nextViewing, nextSelectedId) {
    setEditingField(null);
    setIsUpdatingAll(false);
    setExpandedRowId(null);
    setSession(nextSession);
    setViewingType(nextViewing);
    setSelectedId(nextSelectedId);
  }

  const lockAll = isUpdatingAll;

  /* ---- Confirm modal request (used by every field) ---- */
  const requestConfirm = ({ fieldLabel, prev, next, onCommit }) => {
    setConfirm({
      title: "Save change?",
      subtitle: "Field update",
      fieldLabel, prev, next,
      onConfirm: () => {
        // (mock — would patch state here)
        setConfirm(null);
        onCommit?.();
        setToast(`${fieldLabel} updated`);
      },
    });
  };

  /* ---- Filtered list helpers ---- */
  const filterByOwner = (arr, key = "ownerType") =>
    ownerType === "both" ? arr : arr.filter((x) => x[key] === ownerType);

  const filteredMenus = filterByOwner(MOCK.menus);
  const filteredItems = filterByOwner(Object.values(MOCK.items));
  const filteredModifiers = filterByOwner(Object.values(MOCK.modifiers));
  const filteredOptions = filterByOwner(Object.values(MOCK.options));

  /* ---- Add btn config ---- */
  const addBtnFor = {
    menus: { kind: "menu", label: "New menu" },
    items: { kind: "menuItem", label: "New menu item" },
    modifiers: { kind: "modifier", label: "New modifier" },
    options: { kind: "option", label: "New option" },
  }[session];

  /* ---- Owner type toggle (sessionToggle right side) ---- */
  const toggleOwner = (v) => {
    if (ownerType === "both") setOwnerType(v === "brand" ? "competitor" : "brand");
    else if (ownerType === v) setOwnerType("both");
    else setOwnerType("both");
  };

  /* ---- The sessionToggle top bar (with breadcrumb when in single view) ---- */
  const sessions = [
    { value: "menus", label: "Menus" },
    { value: "items", label: "Items" },
    { value: "modifiers", label: "Modifiers" },
    { value: "options", label: "Options" },
  ];

  const renderTop = () => (
    <div className={`menus_sessionToggle ${editingField || isUpdatingAll ? "" : ""}`}>
      <aside className="menus_sessionToggle_sessions">
        {sessions.map((s) => (
          <button key={s.value}
            className={`menus_sessionToggle_session_btn ${session === s.value && viewingType === "all" ? "active" : ""}`}
            onClick={() => goto(s.value, "all", null)}>
            {s.label}
          </button>
        ))}
      </aside>
      <div className="menus_sessionToggle_middle">
        <Menus_breadcrumb trail={trail} />
      </div>
      <aside className="menus_sessionToggle_ownerTypes">
        {!isUpdatingAll && viewingType === "all" && (
          <_aFrag>
            <button type="button"
              className={`menus_sessionToggle_ownerType_btn ${ownerType === "brand" || ownerType === "both" ? "active" : ""} brand`}
              onClick={() => toggleOwner("brand")} aria-label="Brand">
              <span className="menus_sessionToggle_ownerType_iconWrap">
                <span className="menus_sessionToggle_ownerType_icon" style={{ display: "inline-flex", alignItems: "center", justifyContent: "center", color: "#1d1b18" }}>{I.store}</span>
                {(ownerType === "brand" || ownerType === "both") && <span className="menus_sessionToggle_ownerType_indicator" />}
              </span>
            </button>
            <button type="button"
              className={`menus_sessionToggle_ownerType_btn ${ownerType === "competitor" || ownerType === "both" ? "active" : ""} competitor`}
              onClick={() => toggleOwner("competitor")} aria-label="Competitor">
              <span className="menus_sessionToggle_ownerType_iconWrap">
                <span className="menus_sessionToggle_ownerType_icon" style={{ display: "inline-flex", alignItems: "center", justifyContent: "center", color: "#1d1b18" }}>{I.rival}</span>
                {(ownerType === "competitor" || ownerType === "both") && <span className="menus_sessionToggle_ownerType_indicator" />}
              </span>
            </button>
          </_aFrag>
        )}

        {viewingType === "single" && (
          <div className="menus_FieldUpdateBtn">
            {!isUpdatingAll ? (
              <button className="menus_FieldUpdateBtn_btn" onClick={() => { setEditingField(null); setIsUpdatingAll(true); }}>Update all</button>
            ) : (
              <>
                <button className="menus_FieldUpdateBtn_btn" onClick={() => setIsUpdatingAll(false)}>Cancel</button>
                <button className="menus_FieldUpdateBtn_btn" onClick={() => requestConfirm({
                  fieldLabel: "All fields",
                  prev: "(current)", next: "(your edits)",
                  onCommit: () => { setIsUpdatingAll(false); },
                })}>Confirm</button>
              </>
            )}
          </div>
        )}

        {viewingType === "all" && addBtnFor && (
          <div className="menus_FieldUpdateBtn">
            <button className="menus_FieldUpdateBtn_btn" onClick={() => setShowForm(addBtnFor.kind)}>
              + {addBtnFor.label}
            </button>
          </div>
        )}
      </aside>
    </div>
  );

  /* ---- Body ---- */
  const renderBody = () => {
    // Menus session
    if (session === "menus") {
      if (viewingType === "all") {
        return (
          <div className="menus_session_menu">
            {filteredMenus.length === 0 ? (
              <div className="menus_empty_menus"><p>No menus match the current owner filter.</p></div>
            ) : (
              <Menus_table menus={filteredMenus} expandedId={expandedRowId} setExpandedId={setExpandedRowId}
                onView={(m) => goto("menus", "single", m._id)}
                onUpdate={(m) => { goto("menus", "single", m._id); setIsUpdatingAll(true); }}
              />
            )}
          </div>
        );
      }
      const menu = MOCK.menus.find((m) => m._id === selectedId);
      if (!menu) return <div className="menus_empty_menus"><p>Menu not found.</p></div>;
      return (
        <div className="menus_session_menu">
          <Menu_view_one menu={menu} lockAll={lockAll} editingField={editingField} setEditingField={setEditingField} requestConfirm={requestConfirm}
            onViewItem={(it) => goto("items", "single", it._id)}
            onUpdateItem={(it) => { goto("items", "single", it._id); setIsUpdatingAll(true); }}
            expandedItemId={expandedRowId} setExpandedItemId={setExpandedRowId}
          />
        </div>
      );
    }

    // Items session
    if (session === "items") {
      if (viewingType === "all") {
        const items = filteredItems;
        return (
          <div className="menus_session_menuItem">
            {items.length === 0 ? (<div className="menus_empty_menuItems"><p>No items match the current owner filter.</p></div>) : (
              <Menus_menuItems_table items={items} expandedId={expandedRowId} setExpandedId={setExpandedRowId}
                onView={(it) => goto("items", "single", it._id)}
                onUpdate={(it) => { goto("items", "single", it._id); setIsUpdatingAll(true); }}
              />
            )}
          </div>
        );
      }
      const item = MOCK.items[selectedId];
      if (!item) return <div className="menus_empty_menuItems"><p>Item not found.</p></div>;
      return (
        <div className="menus_session_menuItem">
          <MenuItem_view_one item={item} lockAll={lockAll} editingField={editingField} setEditingField={setEditingField} requestConfirm={requestConfirm} />
        </div>
      );
    }

    // Modifiers session
    if (session === "modifiers") {
      if (viewingType === "all") {
        const ms = filteredModifiers;
        return (
          <div className="menus_session_modifier">
            {ms.length === 0 ? <div className="menus_empty_modifiers"><p>No modifiers match the current owner filter.</p></div> : (
              <Menus_modifier_table modifiers={ms} expandedId={expandedRowId} setExpandedId={setExpandedRowId}
                onView={(m) => goto("modifiers", "single", m._id)}
                onUpdate={(m) => { goto("modifiers", "single", m._id); setIsUpdatingAll(true); }}
              />
            )}
          </div>
        );
      }
      const m = MOCK.modifiers[selectedId];
      if (!m) return <div className="menus_empty_modifiers"><p>Modifier not found.</p></div>;
      return (
        <div className="menus_session_modifier">
          <Modifier_view_one modifier={m} lockAll={lockAll} editingField={editingField} setEditingField={setEditingField} requestConfirm={requestConfirm}
            onViewOption={(o) => goto("options", "single", o._id)} />
        </div>
      );
    }

    // Options session
    if (session === "options") {
      if (viewingType === "all") {
        const os = filteredOptions;
        return (
          <div className="menus_session_option">
            {os.length === 0 ? <div className="menus_empty_modifierOptions"><p>No options match the current owner filter.</p></div> : (
              <Menus_option_table options={os} expandedId={expandedRowId} setExpandedId={setExpandedRowId}
                onView={(o) => goto("options", "single", o._id)}
                onUpdate={(o) => { goto("options", "single", o._id); setIsUpdatingAll(true); }}
              />
            )}
          </div>
        );
      }
      const o = MOCK.options[selectedId];
      if (!o) return <div className="menus_empty_modifierOptions"><p>Option not found.</p></div>;
      return (
        <div className="menus_session_option">
          <Option_view_one option={o} lockAll={lockAll} editingField={editingField} setEditingField={setEditingField} requestConfirm={requestConfirm} />
        </div>
      );
    }

    return null;
  };

  return (
    <div className="menus">
      {renderTop()}
      {renderBody()}

      <Menus_confirmModal
        open={!!confirm}
        title={confirm?.title}
        subtitle={confirm?.subtitle}
        fieldLabel={confirm?.fieldLabel}
        prevValue={confirm?.prev}
        nextValue={confirm?.next}
        onCancel={() => setConfirm(null)}
        onConfirm={() => confirm?.onConfirm?.()}
      />

      {showForm && (
        <Menus_form kind={showForm} onClose={() => setShowForm(null)} onCreate={(payload) => {
          setShowForm(null);
          setToast(`Created ${payload.kind}: ${payload.label}`);
        }} />
      )}

      {toast && <div className="menus_toast">{toast}</div>}
    </div>
  );
}
