## Codebase Conventions (read before touching any file)

| Rule | Detail |
|---|---|
| **Barrels** | Always import from `_index.js` â€” never from individual files |
| **CSS class names** | Root: camelCase (`brandsDetailFiles`). Internal: single underscore `brandsDetailFiles_row`. **Never** double underscore `__`. |
| **Component size** | 100â€“120 lines max per file. Split into child component when exceeded. |
| **Hook pattern** | `useXXX_states` â†’ `useXXX_apiHelpers` â†’ `useXXX_handlers` â†’ `useXXX` (orchestrator). All `useTranslation` calls only in orchestrator. |
| **Props shape** | Orchestrator builds `compProps.XXX_YYY_props`. Components receive via `{...compProps.XXX_YYY_props}`. |
| **No double-underscore** | Global rule. Any `__` in className is a bug. |
| **CSS vars only** | Never hardcode colours in component CSS. Always reference a CSS var from `lightTheme.css` / `darkTheme.css`. |
| **Locale keys** | Every user-visible string via `useTranslation`. Add keys to **all 4** locale files (`en`, `ar`, `ru`, `hy`). |
| **Service logs** | Backend services log `[STARTED]` / `[COMPLETED]` in try/finally when `isDebug`. |
| **Validator return** | `{ isValid, msg, sanitizedData }` â†’ attached to `req.body.sanitizedData`. |

## Task 7 â€” Create `Brands_detail_integration_row.jsx`

Single integration read/edit orchestrator â€” used as a row inside the `inventoryIntegrations` array.

- [ ] **7.1 â€” Create `Brands_detail_integration_row.jsx`**

  ```jsx
  import {
    Brands_detail_integration_payment,
    Brands_detail_integration_credentials,
    Brands_detail_integration_contacts,
    Brands_detail_integration_phones,
    Brands_detail_integration_logins,
  } from "./_brands_childComps.index.js";
  import "../../_styles/brands_detail_integration.css";

  const Brands_detail_integration_row = ({ row, index, isEditing, onRowChange, onRowReplace, onRemove, t }) => {
    const change = (path, value) => {
      const parts = path.split(".");
      const patch = {};
      let cursor = patch;
      for (let i = 0; i < parts.length - 1; i++) {
        cursor[parts[i]] = { ...(row?.[parts[i]] ?? {}) };
        cursor = cursor[parts[i]];
      }
      cursor[parts[parts.length - 1]] = value;
      onRowChange(index, patch);
    };

    const replaceSubArray = (field, next) => onRowChange(index, { [field]: next });

    if (!isEditing) {
      return (
        <div className="brandsDetailIntegration_rowRead">
          <span className="brandsDetailIntegration_rowProvider">
            {row?.provider || t("empty.noValue")}
          </span>
          <span className={`brandsDetailIntegration_badge ${row?.isActive !== false ? "brandsDetailIntegration_badge--active" : "brandsDetailIntegration_badge--inactive"}`}>
            {row?.isActive !== false ? t("badges.active") : t("badges.inactive")}
          </span>
        </div>
      );
    }

    return (
      <div className="brandsDetailIntegration_rowEdit">
        <div className="brandsDetailIntegration_rowHeader">
          <span className="brandsDetailIntegration_rowIndex">#{index + 1}</span>
          <button type="button" className="brandsDetailIntegration_removeBtn" onClick={() => onRemove(index)}>
            {t("actions.removeIntegration")}
          </button>
        </div>
        <div className="brandsDetailIntegration_grid">
          {["provider", "startedAt", "notes"].map((field) => (
            <label className="brandsDetailIntegration_field" key={field}>
              <span className="brandsDetailIntegration_label">{t(`fields.${field}`)}</span>
              <input
                className="brandsDetailIntegration_input"
                type={field === "startedAt" ? "date" : "text"}
                value={field === "startedAt" ? (row?.[field] ? String(row[field]).slice(0, 10) : "") : (row?.[field] ?? "")}
                onChange={(e) => change(field, e.target.value)}
              />
            </label>
          ))}
          <label className="brandsDetailIntegration_toggle">
            <input type="checkbox" checked={row?.isActive !== false} onChange={(e) => change("isActive", e.target.checked)} />
            <span>{t("fields.isActive")}</span>
          </label>
        </div>
        <Brands_detail_integration_payment draft={row} onDraftChange={change} t={t} />
        <Brands_detail_integration_credentials draft={row} onDraftChange={change} t={t} />
        <Brands_detail_integration_contacts draft={row} onDraftChange={change} t={t} />
        <Brands_detail_integration_phones phones={row?.registeredPhones ?? []} onReplace={(next) => replaceSubArray("registeredPhones", next)} t={t} />
        <Brands_detail_integration_logins logins={row?.loginCredentials ?? []} onReplace={(next) => replaceSubArray("loginCredentials", next)} t={t} />
      </div>
    );
  };

  export default Brands_detail_integration_row;
  ```
